 86    def get_transition(self):
 87        if not self.transition:
 88            obj = self.get_object()
 89            impersonated_by = getattr(self.request, 'impersonated_by', None)
 90            self.transition = obj.wfm.get_transition(self.destination_phase, self.request.user, impersonated_by=impersonated_by)

 98    def get_context_data(self, *args, **kwargs):
 99        context = super(ChangeStateView, self).get_context_data(*args, **kwargs)
100        context = self.get_workflow_context(context)
101        transition = self.get_transition()
102        context['transition'] = transition
103        context['last_owner'] = transition.get_last_owner()
104        if transition.command:
105            context['title'], context['message'], context['button_caption'] = self.command_captions[self.destination_phase]
106        else:
107            context['title'] = wgettext('Transition to phase: %(phase)s') % {'phase': transition.destination}
108            context['message'] = self.get_transition_gui_message(transition)
109            context['button_caption'] = transition.caption
110        context['show_user'] = transition.show_owner
111        if not context['show_user']:
112            dest_owner = self.get_destination_owner()
113            if dest_owner:
114                context['dest_owner'] = dest_owner.username
115            else:
116                context['dest_owner'] = wgettext('None')
117        context['show_message'] = True
118        context['cancel_url'] = self.get_cancel_url()
119        return context

167        if owner_id:
168            self.transition.set_owner(get_user_model().objects.get(pk=owner_id))
169        elif self.transition.command in ('', 'release'):
