 23class WFTransitionDescriptor(object):
 24    """
 25    Read-only descriptor for a potential workflow transition, used by the view layer.
 26
 27    Obtained via instance.wfm.get_transition(dest_state, user). Encapsulates
 28    everything a template or view needs to render a transition button and the
 29    corresponding form: caption, owner selection, message requirement, allowed
 30    groups, and whether the transition is currently permitted.
 31
 32    Special command strings ('release', 'delegate', 'assign', 'reject',
 33    'take-ownership', 'suspend', 'resume') are resolved to the appropriate
 34    destination state and owner automatically.
 35
 36    Key properties:
 37    - destination / owner / message  — transition parameters
 38    - caption / is_reject / is_forward / is_free / is_take_ownership  — display hints
 39    - show_owner / require_message  — form field visibility
 40    - allowed() — True if transition_allowed() would pass for current user/state
 41    - execute() — perform the transition (calls obj.wfm.transition())
 42    """

 80    def guess_owner(self):
 81        if self.is_reject:
 82            self._owner = self.get_last_owner()
 83        else:
 84            dom = self.destination_owner_mode
 85            if dom == 'none':
 86                self._owner = None
 87            elif dom == 'user':
 88                self._owner = self.user
 89            elif dom == 'last_owner':
 90                self._owner = self.get_last_owner()
 91            else:
 92                self._owner = None
 93        if self._owner and not self._owner.id in [ow.id for ow in self.obj.get_candidate_users(for_state=self._destination, privileges='ea', active=True)]:
 94            self._owner = None

326    def get_potential_owners(self):
327        user_is_admin = self.wfm.can_admin(self.user)
328        perm = 'e' if not user_is_admin else 'ea'
329        candidates = self.obj.get_candidate_users(self.destination, perm, True)
330        pot_owners = [(user.id, user.wfm_display_str_assign()) for user in candidates]
331        # Remove admin from potential owners for normal users, #592
332        if not user_is_admin:
333            # This shouldn't be needed, admin should have only 'a' privilege on records
334            try:
335                self._remove_users_from_owners((self.obj.wfm_config.admin(), ), pot_owners)
336            except Exception:
337                pass
338        if self.command in ['assign', 'delegate', 'reassign']:
339            # Cannot delegate to myself or to current owner; owner is compulsory
340            self._remove_users_from_owners((self.user, self.obj.current_state.owner), pot_owners)
341        else:
342            if (self.destination_owner_mode in ('assign-optional', 'none')):
343                # Allow empty owner
344                pot_owners.insert(0, (0, '---'))
345        return pot_owners

348    def _remove_users_from_owners(self, users, pot_owners):
349        removable_ids = [user.id for user in users if user]
350        for user_data in pot_owners:
351            if user_data[0] in removable_ids:
352                pot_owners.remove(user_data)

354    @property
355    def severity(self):
356        """Returns 'info', 'warn', or 'error' for button-color mapping.
357
358        Maps to Bootstrap classes: info→btn-outline-primary, warn→btn-warning,
359        error→btn-danger.
360
361        Admin-forced actions (snatch = take-ownership when record is already
362        owned; reassign = delegate on someone else's record) are marked 'error'.
363        """
364        if self.is_forward:
365            return 'info'
366        if self.is_reject:
367            return 'warn'
368        if self.command == 'take-ownership':
369            # snatch: admin taking ownership from an existing owner
370            if self.state and self.state.owner and self.state.owner != self.user:
371                return 'error'
372            return 'info'
373        if self.command == 'resume':
374            return 'info'
375        if self.command == 'delegate':
376            # reassign: admin delegating on someone else's record
377            if self.state and self.state.owner and self.state.owner != self.user:
378                return 'error'
379            return 'warn'
380        if self.command in ('suspend', 'release'):
381            return 'warn'
382        return 'info'

384    @classmethod
385    def get_workflow_transitions(cls, obj, user):
386        """
387        Returns a WorkflowTransitions named tuple for (obj, user).
388
389        Extracted from _WorkflowContextMixin.get_allowed_transitions() and
390        extended with command_transitions (take-ownership, suspend, resume,
391        delegate, release; plus admin snatch / reassign).
392
393        Intended for use in both Django CBV mixins and WorkflowViewSetMixin.
394        """
395        cur_state = obj.current_state
396        phase_transitions = []
397        reject_transition = None
398        command_transitions = []
399
400        if not cur_state:
401            return WorkflowTransitions(phase_transitions, reject_transition, command_transitions)
402
403        prev_state = cur_state.get_previous_state()
404        cfg = cur_state.wfm_state_config()
405        reachable_states = cfg['reachable_states'].keys()
406
407        owner = cur_state.owner
408        is_admin = obj.wfm.can_admin(user)
409        is_owner = (owner == user)
410
411        # When suspended, the only available action is Resume.
412        if cur_state.suspended:
413            if is_owner or is_admin:
414                command_transitions.append(cls(obj, 'resume', user))
415            return WorkflowTransitions(phase_transitions, reject_transition, command_transitions)
416
417        # Build phase_transitions + reject_transition — only for owner.
418        # Admins are not allowed to perform state changes (see transition_allowed);
419        # they use command_transitions (take-ownership, delegate) instead.
420        if is_owner:
421            if cur_state.can_reject and prev_state and prev_state.phase == cur_state.phase:
422                reject_transition = cls(obj, cur_state.phase, user, cur_state.user)
423
424            for dest_state in reachable_states:
425                transition = cls(obj, dest_state, user)
426                if transition.is_reject and prev_state and transition.destination == prev_state.phase:
427                    if cur_state.can_reject:
428                        reject_transition = transition
429                elif not transition.is_reject or cur_state.find_last_state(transition.destination):
430                    phase_transitions.append(transition)
431
432        # Build command_transitions
433
434        if owner is None:
435            if obj.wfm.can_edit(user) or is_admin:
436                command_transitions.append(cls(obj, 'take-ownership', user))
437
438        if owner == user:
439            command_transitions.append(cls(obj, 'suspend', user))
440            if cur_state.can_delegate:
441                command_transitions.append(cls(obj, 'delegate', user))
442            if cur_state.can_release:
443                command_transitions.append(cls(obj, 'release', user))
444
445        if is_admin and owner is not None and owner != user:
446            command_transitions.append(cls(obj, 'take-ownership', user))  # snatch → severity=error
447            command_transitions.append(cls(obj, 'delegate', user))        # reassign → severity=error
448
449        return WorkflowTransitions(phase_transitions, reject_transition, command_transitions)

Special command strings ('release', 'delegate', 'assign', 'reject', 'take-ownership', 'suspend', 'resume') are resolved to the appropriate destination state and owner automatically.
