workflango.contrib.sebastian

Optional drf-sebastian integration for workflango.

Not imported by any core workflango module, and not required to use workflango.drf -- opt in explicitly::

# serializers.py
from workflango.contrib.sebastian import SebastianWorkflowSerializerMixin

class MyModelSerializer(SebastianWorkflowSerializerMixin, serializers.ModelSerializer):
    class Meta:
        model = MyModel
        fields = ['id', 'title', 'current_state']
    class Sebastian:
        list_fields = ['title', 'current_state_for_list']

# views.py
from sebastian.mixins import GUIMixin
from workflango.contrib.sebastian import SebastianWorkflowViewSetMixin

class MyModelViewSet(SebastianWorkflowViewSetMixin, GUIMixin, viewsets.ModelViewSet):
    queryset = MyModel.objects.all()
    serializer_class = MyModelSerializer

Requires djangorestframework and drf-sebastian both installed and configured in the consuming project (drf-sebastian is not published on PyPI -- install it from its own repository). See docs/sebastian-integration.md for the full walkthrough, including the four template fragments this module's template_namespace resolves to (already shipped at workflango/templates/workflango/sebastian/htmx/).

 61class WorkflowActionSerializer(serializers.Serializer):  # pylint: disable=too-few-public-methods
 62    """
 63    Confirmation-form schema for a workflow transition, rendered by
 64    ``sebastian/htmx/confirm.html`` and validated on POST by
 65    ``SebastianWorkflowViewSetMixin.change_state_form``.
 66
 67    The ``user`` field (impersonation) is intentionally excluded -- impersonation is a
 68    privileged API-level feature (see ``WorkflowViewSetMixin.change_state``), not
 69    exposed through this GUI confirm form.
 70    """

 72    owner     = serializers.IntegerField(allow_null=True, required=False, default=None)
 73    message   = serializers.CharField(allow_blank=True, required=False, default='')
 74    suspended = serializers.BooleanField(required=False, default=False)

 77class SebastianWorkflowSerializerMixin(WorkflowSerializerMixin):  # pylint: disable=too-few-public-methods
 78    """
 79    Drop-in replacement for ``workflango.drf.WorkflowSerializerMixin`` that adds a
 80    GUI-only ``current_state_for_list`` column and the ``__can_update`` flag
 81    sebastian's list/detail templates read to decide whether to show an edit link.
 82    """
 83
 84    wf_datetime_format = 'SHORT_DATETIME_FORMAT'
 85    """
 86    Django format-variable name used to format ``State.state_date`` in the
 87    ``current_state_for_list`` badge.  Resolved locale-aware via
 88    ``django.utils.formats.date_format`` — not a strftime pattern.
 89    Override with another recognised name (e.g. ``'DATETIME_FORMAT'``) or a
 90    literal Django template-filter format string (e.g. ``'d/m/Y H:i'``).
 91
 92    Deliberately distinct from ``GUISerializerMixin.datetime_format`` (a strftime
 93    pattern used by drf-sebastian to format DateTimeField values in the API
 94    response).  Setting this attribute does not affect those field renderings.
 95    """
 96
 97    def to_representation(self, instance):
 98        ret = super().to_representation(instance)
 99        request = self.context.get('request')
100        if getattr(request, 'sebastian_gui', False):
101            state = instance.wfm_state
102            if state:
103                impersonated_by = getattr(request, 'impersonated_by', None)
104                ret['__can_update'] = state.owned_by(request.user, impersonated_by) and not state.suspended
105            # No __can_update when unmanaged: template falls back to view.can_update() (True)
106        return ret
107
108    @gui_field(wgettext_lazy('State'))
109    def current_state_for_list(self, obj):
110        """Rendered as a Sebastian.list_fields column; never in the JSON API response."""
111        from django.utils.formats import date_format
112        from django.utils.html import format_html, conditional_escape
113        from django.utils.safestring import mark_safe
114        from django.utils.timezone import localtime
115        state = obj.wfm_state
116        if not state:
117            return '—'
118        date_str = date_format(localtime(state.state_date), self.wf_datetime_format) if state.state_date else ''
119        owner_str = str(state.owner) if state.owner else '—'
120        icons = ''
121        if state.suspended:
122            icons += '<i class="bi bi-hourglass-split ms-1 text-warning" title="Suspended"></i>'
123        if state.message:
124            icons += f'<i class="bi bi-sticky ms-1 text-muted" title="{conditional_escape(state.message)}"></i>'
125        return format_html(
126            '<span class="badge bg-secondary">{}</span>'
127            ' <span class="ms-1">{}</span>'
128            ' <small class="text-muted ms-1">{}</small>'
129            '{}',
130            state.phase or '—', owner_str, date_str, mark_safe(icons),
131        )

134class SebastianWorkflowViewSetMixin(WorkflowViewSetMixin):
135    """
136    Drop-in replacement for ``workflango.drf.WorkflowViewSetMixin`` that adds the
137    GUI permission hooks and actions sebastian's ``GUIMixin``/renderer/router expect
138    by naming convention (``can_update``/``can_delete``/``get_workflow_transitions``,
139    the ``template_namespace`` used to resolve ``workflango/sebastian/{pack}/...``
140    templates ahead of sebastian's own, and the ``history``/``change_state_form``
141    GUI-flavored actions).
142
143    Place before sebastian's ``GUIMixin`` in the MRO so these override its defaults::
144
145        class MyModelViewSet(SebastianWorkflowViewSetMixin, GUIMixin, viewsets.ModelViewSet):
146            ...
147    """
148
149    template_namespace = 'workflango'
150
151    # ------------------------------------------------------------------
152    # GUI permission hooks (override GUIMixin defaults)
153    # ------------------------------------------------------------------
154
155    def can_update(self):
156        """Returns True only when the current user owns the workflow instance.
157
158        In list context (_sebastian_obj not set) returns True so the per-row
159        edit link is shown; ownership is enforced when the form actually loads.
160        """
161        obj = getattr(self, '_sebastian_obj', None)
162        if obj is None:
163            return True
164        if not obj.wfm_state:
165            return True   # object not yet under workflow management
166        impersonated_by = getattr(self.request, 'impersonated_by', None)
167        return obj.wfm.is_owner(self.request.user, impersonated_by) and not obj.wfm_state.suspended
168
169    def can_delete(self):
170        """Workflow-managed objects cannot be deleted via the GUI."""
171        return False
172
173    def get_workflow_transitions(self, instance):
174        """
175        Returns WorkflowTransitions(phase_transitions, reject_transition, command_transitions)
176        for the given instance and the current request user.
177
178        sebastian's renderer duck-types on this method and, if present, injects the
179        result into the template context as ``workflow_transitions`` so the
180        ``workflango/sebastian/htmx/detail.html`` template can render the action buttons.
181        """
182        return WFTransitionDescriptor.get_workflow_transitions(instance, self.request.user)
183
184    # ------------------------------------------------------------------
185    # Actions
186    # ------------------------------------------------------------------
187
188    @action(detail=True, methods=['get'])
189    def history(self, request, pk=None, **kwargs):  # noqa: ARG002
190        """Returns all State records for this instance (GUI-friendly alias for workflow_history)."""
191        instance = self.get_object()
192        self._sebastian_obj = instance
193        states = instance.wfm.get_states().select_related('owner', 'user', 'impersonated_by')
194        page = self.paginate_queryset(states)
195        if page is not None:
196            serializer = self.get_state_serializer(page, many=True)
197            return self.get_paginated_response(serializer.data)
198        serializer = self.get_state_serializer(states, many=True)
199        return Response(serializer.data)
200
201    history.gui_config = {
202        'label': wgettext_lazy('History'),
203        'icon': 'clock-history',
204        'position': 'both',
205    }
206
207    @action(detail=True, methods=['get', 'post'])
208    def change_state_form(self, request, pk=None, **__):  # noqa: ARG002
209        """
210        GET: returns the inline confirmation form for a workflow transition.
211             Accepts ``?phase=<phase>`` query parameter.
212        POST: performs the transition and returns the instance detail data so
213              HTMX can reload the full detail page into ``#sebastian-content``.
214
215        Both GET and POST render through the workflango confirm template when
216        present, falling back to the base Sebastian confirm template.
217        """
218        instance = self.get_object()
219        self._sebastian_obj = instance
220
221        if request.method == 'GET':
222            phase = request.query_params.get('phase')
223            if not phase:
224                raise DRFValidationError({'phase': 'Required.'})
225
226            transition = WFTransitionDescriptor(instance, phase, request.user)
227            owner_choices = transition.get_potential_owners() if transition.show_owner else []
228            default_owner = transition.get_default_owner() if transition.show_owner else None
229            severity_to_style = {'info': 'primary', 'warn': 'warning', 'error': 'danger'}
230
231            pre_check_blocked = not transition.allowed()
232            pre_check_errors = (
233                [getattr(transition, 'error_msg', 'Transition not allowed.')]
234                if pre_check_blocked else []
235            )
236
237            response_data = {
238                'action': 'confirm',
239                'confirm_prompt': transition.caption,
240                'confirm_style': severity_to_style.get(transition.severity, 'primary'),
241                'action_url': request.path,
242                'confirm_serializer': WorkflowActionSerializer(initial={
243                    'phase': phase,
244                    'suspended': transition.is_suspend,
245                }),
246                'owner_choices': owner_choices,
247                'default_owner_id': default_owner.pk if default_owner else None,
248                'show_owner': transition.show_owner,
249                'require_message': transition.require_message,
250                'phase': phase,
251                'suspended': transition.is_suspend,
252                'pre_check_blocked': pre_check_blocked,
253                'warnings': instance.wfm.warnings,
254                'infos': instance.wfm.infos,
255            }
256            if pre_check_errors:
257                response_data['form_errors'] = {'non_field_errors': pre_check_errors}
258            return Response(response_data)
259
260        # POST: perform the transition
261        input_ser = WorkflowActionSerializer(data=request.data)
262        input_ser.is_valid(raise_exception=True)
263        data = input_ser.validated_data
264
265        self.check_wf_permission(instance, request.user)
266
267        # Resolve command strings ('suspend', 'resume', 'release', 'take-ownership', ...)
268        # to the real destination phase and suspended flag via WFTransitionDescriptor.
269        phase      = data['phase']
270        transition = WFTransitionDescriptor(instance, phase, request.user)
271        destination = transition.destination
272        suspended   = transition.is_suspend
273
274        form_errors: dict = {}
275
276        if data['owner']:
277            try:
278                owner = get_user_model().objects.get(pk=data['owner'], is_active=True)
279            except ObjectDoesNotExist:
280                form_errors['owner'] = [f"User {data['owner']} not found."]
281                owner = None
282        else:
283            owner = transition.owner
284
285        if not form_errors:
286            impersonated_by = getattr(request, 'impersonated_by', None)
287            try:
288                instance.wfm.transition(
289                    request.user,
290                    destination,
291                    owner,
292                    message=data['message'],
293                    suspended=suspended,
294                    impersonated_by=impersonated_by,
295                )
296            except (TransitionNotAllowed, ValidationError) as e:
297                form_errors['non_field_errors'] = [get_exception_error_msg(e)]
298
299        if form_errors:
300            owner_choices = transition.get_potential_owners() if transition.show_owner else []
301            default_owner = transition.get_default_owner() if transition.show_owner else None
302            severity_to_style = {'info': 'primary', 'warn': 'warning', 'error': 'danger'}
303            response = Response({
304                'action':          'confirm',
305                'confirm_prompt':  transition.caption,
306                'confirm_style':   severity_to_style.get(transition.severity, 'primary'),
307                'action_url':      request.path,
308                'owner_choices':   owner_choices,
309                'default_owner_id': default_owner.pk if default_owner else None,
310                'show_owner':      transition.show_owner,
311                'require_message': transition.require_message,
312                'phase':           phase,
313                'suspended':       transition.is_suspend,
314                'form_errors':     form_errors,
315            }, status=400)
316            response['X-Sebastian-Form-Error'] = '1'
317            response['HX-Retarget'] = '#wf-confirm-panel'
318            response['HX-Reswap'] = 'innerHTML'
319            return response
320
321        instance.refresh_from_db()
322        serializer = self.get_serializer(instance)
323        return Response(serializer.data)
324
325    change_state_form.gui_url = True  # register in GUIRouter without adding to action buttons
