workflango.drf

Django REST Framework integration for workflango.

Requires djangorestframework to be installed in the consuming project.

Typical usage::

# serializers.py
from workflango.drf import WorkflowSerializerMixin

class MyModelSerializer(WorkflowSerializerMixin, serializers.ModelSerializer):
    class Meta:
        model = MyModel
        fields = ['id', 'title', 'current_state']

# views.py
from workflango.drf import WorkflowViewSetMixin, WorkflowFilterBackend

class MyModelViewSet(WorkflowViewSetMixin, ModelViewSet):
    queryset = MyModel.objects.all()
    serializer_class = MyModelSerializer
    filter_backends = [WorkflowFilterBackend]

    def get_queryset(self):
        # Scope results to the effective user when ?as_user= is specified
        acting_user, _ = self.get_effective_user(self.request)
        return MyModel.objects.filter(wfm_state__owner=acting_user)

140    ViewSet mixin that adds standard workflow actions to any ModelViewSet.
141
142    Actions added:
143    - GET  ``/{pk}/workflow_history/``  — ordered list of all State records
144    - POST ``/{pk}/change_state/``      — perform a workflow transition
145
146    Impersonation:
147    - POST ``change_state`` accepts an optional ``user`` field in the body.
148      If ``user != request.user`` and the caller is allowed to impersonate,
149      the transition is recorded as ``user`` acting, ``request.user`` as ``impersonated_by``.

151      The consuming ViewSet can call ``get_effective_user(request)`` in ``get_queryset()``
152      to scope results accordingly.

156        WORKFLANGO_ALLOW_IMPERSONATE = True  # in settings.py

158    Override ``get_impersonable_users(user)`` to customise who can be impersonated by whom.
159

162        class MyModelViewSet(WorkflowViewSetMixin, ModelViewSet):

193    def resolve_acting_user(self, request, user_id=None):
194        """
195        Resolves the acting user and impersonation context for a request.
196
197        Returns ``(acting_user, impersonated_by)`` where:
198        - ``acting_user`` is who the operation will be attributed to
199        - ``impersonated_by`` is the real caller (or None if no impersonation)
200
201        Raises ``PermissionDenied`` if ``user_id`` is specified but the caller
202        is not allowed to impersonate that user, or if impersonation is disabled
203        globally (``WORKFLANGO_ALLOW_IMPERSONATE`` not True in settings).
204        """
205        if not user_id or user_id == request.user.pk:
206            return request.user, None
207
208        if not getattr(settings, 'WF_ALLOW_IMPERSONATE', False):
209            raise PermissionDenied(wgettext("Impersonation is not enabled (WF_ALLOW_IMPERSONATE)."))
210
211        try:
212            target_user = get_user_model().objects.get(pk=user_id, is_active=True)
213        except ObjectDoesNotExist as exc:
214            raise DRFValidationError({
215                'user': wgettext("User %(user_id)s not found or not active.") % {'user_id': user_id},
216            }) from exc
217
218        if not self.get_impersonable_users(request.user).filter(pk=user_id).exists():
219            raise PermissionDenied(wgettext("Not authorized to act as %(user)s.") % {'user': target_user})
220
221        return target_user, request.user
222
223    def get_effective_user(self, request):
224        """
225        Resolves the effective user from the ``?as_user=<id>`` query parameter.
226
227        Returns ``(acting_user, impersonated_by)`` — same contract as
228        ``resolve_acting_user``. Intended for use in ``get_queryset()`` to scope
229        list results to what the acting user would see::
230
231            def get_queryset(self):
232                acting_user, _ = self.get_effective_user(self.request)
233                return MyModel.objects.filter(wfm_state__owner=acting_user)
234        """
235        as_user_id = request.query_params.get('as_user')
236        if not as_user_id:
237            return request.user, None
238        try:
239            as_user_id = int(as_user_id)
240        except (ValueError, TypeError) as exc:
241            raise DRFValidationError({'as_user': wgettext("Invalid value: expected an integer.")}) from exc
242        return self.resolve_acting_user(request, as_user_id)

248    def check_wf_permission(self, instance, acting_user):
249        """
250        Raises ``PermissionDenied`` if ``acting_user`` is not allowed to perform
251        a workflow transition. Skipped for unmanaged objects (first transition).
252
253        Allowed if any of:
254        - acting_user is the current owner (with matching impersonation context)
255        - acting_user is the recorded owner acting outside any impersonation
256          (allows reclaiming own identity — transition_allowed handles the rest)
257        - acting_user is an admin for this instance
258        - the instance has no owner and acting_user can edit in the current phase
259          (i.e. they are eligible to take ownership)
260        """
261        if not instance.wfm_state:
262            return
263        impersonated_by = getattr(self.request, 'impersonated_by', None)
264        if instance.wfm.is_owner(acting_user, impersonated_by):
265            return
266        if instance.wfm_state.owner == acting_user:
267            return
268        if instance.wfm.can_admin(acting_user):
269            return
270        if instance.wfm_state.owner is None and instance.wfm.can_edit(acting_user):
271            return
272        raise PermissionDenied(wgettext("Access denied: user is neither the owner nor an administrator."))

290    @action(detail=True, methods=['post'])
291    def change_state(self, request, pk=None):  # noqa: ARG002
292        """
293        Perform a workflow transition.
294
295        Request body::
296
297            {
298                "phase":     "nuova_fase",
299                "owner":     42,     // user id; null or omitted = no owner
300                "user":      15,     // optional: act as this user (impersonation)
301                "message":   "...",  // optional
302                "suspended": false   // optional
303            }
304
305        If ``user`` differs from the caller, the caller is saved in
306        ``State.impersonated_by`` and ``user`` becomes the recorded actor.
307
308        Returns the new State (201) on success.
309        Returns 400 on validation / transition failure.
310        Returns 403 if not owner/admin, or impersonation is not allowed.
311        """
312        instance = self.get_object()
313
314        input_ser = _ChangeStateInputSerializer(data=request.data)
315        input_ser.is_valid(raise_exception=True)
316        data = input_ser.validated_data
317
318        acting_user, impersonated_by = self.resolve_acting_user(request, data['user'])
319        self.check_wf_permission(instance, acting_user)
320
321        owner = None
322        if data['owner']:
323            try:
324                owner = get_user_model().objects.get(pk=data['owner'], is_active=True)
325            except ObjectDoesNotExist as exc:
326                raise DRFValidationError({
327                    'owner': wgettext("User %(user_id)s not found.") % {'user_id': data['owner']},
328                }) from exc
329
330        try:
331            new_state = instance.wfm.transition(
332                acting_user,
333                data['phase'],
334                owner,
335                message=data['message'],
336                suspended=data['suspended'],
337                impersonated_by=impersonated_by,
338            )
339        except (TransitionNotAllowed, ValidationError) as e:
340            raise DRFValidationError({'detail': get_exception_error_msg(e)}) from e
341
342        serializer = self.get_state_serializer(new_state)
343        return Response(serializer.data, status=status.HTTP_201_CREATED)

345    @action(detail=True, methods=['post'])
346    def mark_read(self, request, pk=None):  # noqa: ARG002
347        """
348        Set or clear the ``unread`` flag on the current state.
349
350        Only the current owner (or an impersonated user acting as owner) may call
351        this endpoint.  Intended for two patterns:
352
353        1. **Auto mark-read**: the detail template fires a delayed POST after a few
354           seconds with ``{"read": true}`` (or no body, default is ``true``).
355        2. **Toggle button**: the client sends the desired state explicitly::
356
357               {"read": true}   # mark as read   → unread becomes False
358               {"read": false}  # mark as unread  → unread becomes True
359
360        Returns ``{"unread": <bool>}`` so the client can update its UI without a
361        full page reload.
362
363        Returns 403 if the caller is not the current owner.
364        Returns 404 if the instance has no active state yet.
365        """
366        instance = self.get_object()
367        current_state = instance.wfm_state
368        if not current_state:
369            raise PermissionDenied(wgettext("The object does not have an active state yet."))
370
371        acting_user, _ = self.get_effective_user(request)
372        if current_state.owner_id != acting_user.pk:
373            raise PermissionDenied(wgettext("Only the current owner can change the read state."))
374
375        input_ser = _MarkReadInputSerializer(data=request.data)
376        input_ser.is_valid(raise_exception=True)
377        current_state.unread = not input_ser.validated_data['read']
378        current_state.save(update_fields=['unread'])
379        return Response({'unread': current_state.unread})

ViewSet mixin that adds standard workflow actions to any ModelViewSet.

Actions added:

  • GET /{pk}/workflow_history/ — ordered list of all State records
  • POST /{pk}/change_state/ — perform a workflow transition

Impersonation:

  • POST change_state accepts an optional user field in the body. If user != request.user and the caller is allowed to impersonate, the transition is recorded as user acting, request.user as impersonated_by.
  • GET endpoints accept ?as_user=<id> to resolve the effective user. The consuming ViewSet can call get_effective_user(request) in get_queryset() to scope results accordingly.

Impersonation is disabled by default. Enable globally with::

WORKFLANGO_ALLOW_IMPERSONATE = True  # in settings.py

Override get_impersonable_users(user) to customise who can be impersonated by whom.
