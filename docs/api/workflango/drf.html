workflango.drf

Django REST Framework integration for workflango.

Requires djangorestframework to be installed in the consuming project.

Typical usage::

# serializers.py
from workflango.drf import WorkflowSerializerMixin

class MyModelSerializer(WorkflowSerializerMixin, serializers.ModelSerializer):
    class Meta:
        model = MyModel
        fields = ['id', 'title', 'current_state']

# views.py
from workflango.drf import WorkflowViewSetMixin, WorkflowFilterBackend

class MyModelViewSet(WorkflowViewSetMixin, ModelViewSet):
    queryset = MyModel.objects.all()
    serializer_class = MyModelSerializer
    filter_backends = [WorkflowFilterBackend]

    def get_queryset(self):
        # Scope results to the effective user when ?as_user= is specified
        acting_user, _ = self.get_effective_user(self.request)
        return MyModel.objects.filter(wfm_state__owner=acting_user)

113class WorkflowSerializerMixin(serializers.Serializer):  # pylint: disable=too-few-public-methods
114    """
115    Serializer mixin that adds a read-only ``current_state`` nested field and a
116    GUI-only ``current_state_for_list`` column to any WorkflowModel serializer.
117
118    Usage::
119
120        class MyWorkflowModelSerializer(WorkflowSerializerMixin, serializers.ModelSerializer):
121            class Meta:
122                model = MyWorkflowModel
123                fields = ['id', 'title', 'current_state']

126    consuming serializer.
127
128    ``current_state_for_list`` is a ``@gui_field`` method — it renders the workflow
129    state as a formatted HTML badge for use in ``Sebastian.list_fields``. It is
130    never included in the JSON API response.

133    current_state = StateSerializer(read_only=True)
134    datetime_format = '%d/%m/%Y %H:%M'

136    def to_representation(self, instance):
137        ret = super().to_representation(instance)
138        request = self.context.get('request')
139        if getattr(request, 'sebastian_gui', False):
140            state = instance.wfm_state
141            if state:
142                ret['__can_update'] = (state.owner == request.user) and not state.suspended
143            # No __can_update when unmanaged: template falls back to view.can_update (True)
144        return ret

146    @gui_field(wgettext_lazy('State'))
147    def current_state_for_list(self, obj):
148        from django.utils.html import format_html, conditional_escape
149        from django.utils.safestring import mark_safe
150        from django.utils.timezone import localtime
151        state = obj.wfm_state
152        if not state:
153            return '—'
154        date_str = localtime(state.state_date).strftime(self.datetime_format) if state.state_date else ''
155        owner_str = str(state.owner) if state.owner else '—'
156        icons = ''
157        if state.suspended:
158            icons += '<i class="bi bi-hourglass-split ms-1 text-warning" title="Sospeso"></i>'
159        if state.message:
160            icons += f'<i class="bi bi-sticky ms-1 text-muted" title="{conditional_escape(state.message)}"></i>'
161        return format_html(
162            '<span class="badge bg-secondary">{}</span>'
163            ' <span class="ms-1">{}</span>'
164            ' <small class="text-muted ms-1">{}</small>'
165            '{}',
166            state.phase or '—', owner_str, date_str, mark_safe(icons),
167        )

202class WorkflowViewSetMixin:
203    """
204    ViewSet mixin that adds standard workflow actions to any ModelViewSet.
205
206    When used alongside a Sebastian GUIMixin, sets ``template_namespace = 'workflango'``
207    so the Sebastian renderer picks up templates from
208    ``workflango/sebastian/{pack}/`` instead of the default ``sebastian/{pack}/``.
209
210    Actions added:
211    - GET  ``/{pk}/workflow_history/``  — ordered list of all State records
212    - POST ``/{pk}/change_state/``      — perform a workflow transition
213
214    Impersonation:
215    - POST ``change_state`` accepts an optional ``user`` field in the body.
216      If ``user != request.user`` and the caller is allowed to impersonate,
217      the transition is recorded as ``user`` acting, ``request.user`` as ``impersonated_by``.
218    - GET endpoints accept ``?as_user=<id>`` to resolve the effective user.
219      The consuming ViewSet can call ``get_effective_user(request)`` in ``get_queryset()``
220      to scope results accordingly.
221
222    Impersonation is disabled by default. Enable globally with::
223
224        WORKFLANGO_ALLOW_IMPERSONATE = True  # in settings.py
225
226    Override ``get_impersonable_users(user)`` to customise who can be impersonated by whom.
227
228    Usage::
229
230        class MyModelViewSet(WorkflowViewSetMixin, ModelViewSet):
231            queryset = MyModel.objects.all()
232            serializer_class = MyModelSerializer
233            filter_backends = [WorkflowFilterBackend]

296    def resolve_acting_user(self, request, user_id=None):
297        """
298        Resolves the acting user and impersonation context for a request.
299
300        Returns ``(acting_user, impersonated_by)`` where:
301        - ``acting_user`` is who the operation will be attributed to
302        - ``impersonated_by`` is the real caller (or None if no impersonation)
303
304        Raises ``PermissionDenied`` if ``user_id`` is specified but the caller
305        is not allowed to impersonate that user, or if impersonation is disabled
306        globally (``WORKFLANGO_ALLOW_IMPERSONATE`` not True in settings).
307        """
308        if not user_id or user_id == request.user.pk:
309            return request.user, None
310
311        if not getattr(settings, 'WORKFLANGO_ALLOW_IMPERSONATE', False):
312            raise PermissionDenied("Impersonazione non abilitata (WORKFLANGO_ALLOW_IMPERSONATE).")
313
314        try:
315            target_user = get_user_model().objects.get(pk=user_id, is_active=True)
316        except ObjectDoesNotExist as exc:
317            raise DRFValidationError({'user': f'Utente {user_id} non trovato o non attivo.'}) from exc
318
319        if not self.get_impersonable_users(request.user).filter(pk=user_id).exists():
320            raise PermissionDenied(f'Non autorizzato a operare come {target_user}.')
321
322        return target_user, request.user

324    def get_effective_user(self, request):
325        """
326        Resolves the effective user from the ``?as_user=<id>`` query parameter.
327
328        Returns ``(acting_user, impersonated_by)`` — same contract as
329        ``resolve_acting_user``. Intended for use in ``get_queryset()`` to scope
330        list results to what the acting user would see::
331
332            def get_queryset(self):
333                acting_user, _ = self.get_effective_user(self.request)
334                return MyModel.objects.filter(wfm_state__owner=acting_user)
335        """
336        as_user_id = request.query_params.get('as_user')
337        if not as_user_id:
338            return request.user, None
339        try:
340            as_user_id = int(as_user_id)
341        except (ValueError, TypeError) as exc:
342            raise DRFValidationError({'as_user': 'Valore non valido: atteso intero.'}) from exc
343        return self.resolve_acting_user(request, as_user_id)

349    def check_wf_permission(self, instance, acting_user):
350        """
351        Raises ``PermissionDenied`` if ``acting_user`` is not allowed to perform
352        a workflow transition. Skipped for unmanaged objects (first transition).
353
354        Allowed if any of:
355        - acting_user is the current owner
356        - acting_user is an admin for this instance
357        - the instance has no owner and acting_user can edit in the current phase
358          (i.e. they are eligible to take ownership)
359        """
360        if not instance.wfm_state:
361            return
362        if instance.wfm.is_owner(acting_user):
363            return
364        if instance.wfm.can_admin(acting_user):
365            return
366        if instance.wfm_state.owner is None and instance.wfm.can_edit(acting_user):
367            return
368        raise PermissionDenied(wgettext("Access denied: user is neither the owner nor an administrator."))

405    @action(detail=True, methods=['get', 'post'])
406    def change_state_form(self, request, pk=None, **__):  # noqa: ARG002
407        """
408        GET: returns the inline confirmation form for a workflow transition.
409             Accepts ``?phase=<phase>`` query parameter.
410        POST: performs the transition and returns the instance detail data so
411              HTMX can reload the full detail page into ``#sebastian-content``.
412
413        Both GET and POST render through the workflango confirm template when
414        present, falling back to the base Sebastian confirm template.
415        """
416        instance = self.get_object()
417        self._sebastian_obj = instance
418
419        from .wf_transition import WFTransitionDescriptor
420
421        if request.method == 'GET':
422            phase = request.query_params.get('phase')
423            if not phase:
424                raise DRFValidationError({'phase': 'Required.'})
425
426            transition = WFTransitionDescriptor(instance, phase, request.user)
427            owner_choices = transition.get_potential_owners() if transition.show_owner else []
428            default_owner = transition.get_default_owner() if transition.show_owner else None
429            severity_to_style = {'info': 'primary', 'warn': 'warning', 'error': 'danger'}
430
431            pre_check_blocked = not transition.allowed()
432            pre_check_errors = (
433                [getattr(transition, 'error_msg', 'Transizione non consentita.')]
434                if pre_check_blocked else []
435            )
436
437            response_data = {
438                'action': 'confirm',
439                'confirm_prompt': transition.caption,
440                'confirm_style': severity_to_style.get(transition.severity, 'primary'),
441                'action_url': request.path,
442                'confirm_serializer': WorkflowActionSerializer(initial={
443                    'phase': phase,
444                    'suspended': transition.is_suspend,
445                }),
446                'owner_choices': owner_choices,
447                'default_owner_id': default_owner.pk if default_owner else None,
448                'show_owner': transition.show_owner,
449                'require_message': transition.require_message,
450                'phase': phase,
451                'suspended': transition.is_suspend,
452                'pre_check_blocked': pre_check_blocked,
453                'warnings': instance.wfm.warnings,
454                'infos': instance.wfm.infos,
455            }
456            if pre_check_errors:
457                response_data['form_errors'] = {'non_field_errors': pre_check_errors}
458            return Response(response_data)
459
460        # POST: perform the transition
461        input_ser = WorkflowActionSerializer(data=request.data)
462        input_ser.is_valid(raise_exception=True)
463        data = input_ser.validated_data
464
465        self.check_wf_permission(instance, request.user)
466
467        # Resolve command strings ('suspend', 'resume', 'release', 'take-ownership', ...)
468        # to the real destination phase and suspended flag via WFTransitionDescriptor.
469        phase      = data['phase']
470        transition = WFTransitionDescriptor(instance, phase, request.user)
471        destination = transition.destination
472        suspended   = transition.is_suspend
473
474        form_errors: dict = {}
475
476        if data['owner']:
477            try:
478                owner = get_user_model().objects.get(pk=data['owner'], is_active=True)
479            except ObjectDoesNotExist:
480                form_errors['owner'] = [f"Utente {data['owner']} non trovato."]
481                owner = None
482        else:
483            owner = transition.owner
484
485        if not form_errors:
486            impersonated_by = getattr(request, 'impersonated_by', None)
487            try:
488                instance.wfm.transition(
489                    request.user,
490                    destination,
491                    owner,
492                    message=data['message'],
493                    suspended=suspended,
494                    impersonated_by=impersonated_by,
495                )
496            except (TransitionNotAllowed, ValidationError) as e:
497                form_errors['non_field_errors'] = [get_exception_error_msg(e)]
498
499        if form_errors:
500            owner_choices = transition.get_potential_owners() if transition.show_owner else []
501            default_owner = transition.get_default_owner() if transition.show_owner else None
502            severity_to_style = {'info': 'primary', 'warn': 'warning', 'error': 'danger'}
503            response = Response({
504                'action':          'confirm',
505                'confirm_prompt':  transition.caption,
506                'confirm_style':   severity_to_style.get(transition.severity, 'primary'),
507                'action_url':      request.path,
508                'owner_choices':   owner_choices,
509                'default_owner_id': default_owner.pk if default_owner else None,
510                'show_owner':      transition.show_owner,
511                'require_message': transition.require_message,
512                'phase':           phase,
513                'suspended':       transition.is_suspend,
514                'form_errors':     form_errors,
515            }, status=400)
516            response['X-Sebastian-Form-Error'] = '1'
517            response['HX-Retarget'] = '#wf-confirm-panel'
518            response['HX-Reswap'] = 'innerHTML'
519            return response
520
521        instance.refresh_from_db()
522        serializer = self.get_serializer(instance)
523        return Response(serializer.data)

527    @action(detail=True, methods=['post'])
528    def change_state(self, request, pk=None):  # noqa: ARG002
529        """
530        Perform a workflow transition.
531
532        Request body::
533
534            {
535                "phase":     "nuova_fase",
536                "owner":     42,     // user id; null or omitted = no owner
537                "user":      15,     // optional: act as this user (impersonation)
538                "message":   "...",  // optional
539                "suspended": false   // optional
540            }
541
542        If ``user`` differs from the caller, the caller is saved in
543        ``State.impersonated_by`` and ``user`` becomes the recorded actor.
544
545        Returns the new State (201) on success.
546        Returns 400 on validation / transition failure.
547        Returns 403 if not owner/admin, or impersonation is not allowed.
548        """
549        instance = self.get_object()
550
551        input_ser = _ChangeStateInputSerializer(data=request.data)
552        input_ser.is_valid(raise_exception=True)
553        data = input_ser.validated_data
554
555        acting_user, impersonated_by = self.resolve_acting_user(request, data['user'])
556        self.check_wf_permission(instance, acting_user)
557
558        owner = None
559        if data['owner']:
560            try:
561                owner = get_user_model().objects.get(pk=data['owner'], is_active=True)
562            except ObjectDoesNotExist as exc:
563                raise DRFValidationError({'owner': f"Utente {data['owner']} non trovato."}) from exc
564
565        try:
566            new_state = instance.wfm.transition(
567                acting_user,
568                data['phase'],
