 69class WorkflowFilterForm(forms.Form):
 70    """
 71    Every field is `required=False`: this is a search/filter form, not a data-entry
 72    form -- an unfilled field must mean "don't filter on this", not "please pick
 73    something before submitting". Concretely: search_wf_history's first choice is
 74    `('', 'Current')` (an empty *value*, not an empty *selection*); with the field
 75    left at Django's default `required=True`, the browser's HTML5 validation refuses
 76    to submit an empty-valued selection at all, blocking the form even though a
 77    perfectly meaningful choice ("Current") is already selected. Same issue on every
 78    other field here (blank Select/date/text left untouched by the user). Leaving
 79    search_wf_owner/search_wf_phase blank means "no filter", which is a
 80    distinct, still-selectable choice from explicitly picking e.g. owner "None"
 81    (`none` in USER_CHOICES, meaning "owner is null") -- required=False doesn't change
 82    that distinction, it only stops the browser from forcing a choice.
 83    """
